% ---------- PROBLEM ----------
\begin{frame}{Problem Statement}
\begin{block}{Clinical question}
A single base change in \textit{BRCA1} (e.g.\ \texttt{chr17:43119628 G}) can be harmless or sharply raise cancer risk. Telling the two apart needs population data, curation (ACMG) and a fast model-based second opinion.
\end{block}
\begin{itemize}
\item Curated rules are reliable but slow for rare/unseen variants.
\item Classical scores (SIFT, PolyPhen-2, CADD) use hand-built features.
\item DNA language models learn constraint from sequence --- but live in notebooks, not clinics.
\end{itemize}
\begin{block}{Validation question}
What must a deployed genomic model service reveal before its output can be read as proof?
\end{block}
\begin{itemize}
\item A model score is not a clinical label: the move from score to reading is a software contract.
\item Contracts fail in ways models do not: a wrong assembly yields a confident answer about a different base.
\end{itemize}
\end{frame}

% ---------- OBJECTIVES ----------
\begin{frame}{Objectives}
\begin{enumerate}
\item Specify the scoring contract: window construction, the four supported edit operations, the score difference, the threshold rule and the confidence transform.
\item Characterize the calibration data: reproduce the assay composition and reveal that the 500-row slice used for the deployed threshold is class-enriched and position-localized.
\item Compute the first reproducible reference envelope for that slice, with bootstrap confidence intervals.
\item Verify the windowing and assembly contracts: deployed window is calibration window +1 base at the 3' end; deployed threshold was fitted on GRCh37 while the service answers for GRCh38.
\item Execute the request conformance suite and report the reproduced defects.
\end{enumerate}
\end{frame}

% ---------- METHODOLOGY SCORING ----------
\begin{frame}{Methodology: Windowing and Scoring Contract}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\small Let $p$ be the 1-based coordinate of the substituted, inserted or first deleted base; $n$ the deletion length and $q$ the inserted string.
\begin{equation*}
s{=}\max(0,c-1-h),\; e{=}c-1+h+1
\end{equation*}
with window $W{=}8192$, half width $h{=}4096$; $c{=}p$ for SNV/insertion, $c{=}p+\lfloor(n-1)/2\rfloor$ for deletion/delins. Interior windows are $8193$ bases; the variant sits at offset $r{=}p-1-s{=}4096$.
\begin{align*}
w^{\mathrm{SNV}} &= w_{0:r} + a + w_{r+1:},\\
w^{\mathrm{DEL}} &= w_{0:r} + w_{r+n:},\\
w^{\mathrm{INS}} &= w_{0:r+1} + q + w_{r+1:},\\
w^{\mathrm{DELINS}} &= w_{0:r} + q + w_{r+n:}.
\end{align*}
\textit{Edits} that change length shift the downstream context.
\end{column}

\begin{frame}{Methodology: Score Difference, Threshold and Confidence}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\small $S(\cdot)$ is the mean per-base log-likelihood returned by \texttt{model.score\_sequences} (nats per base pair).
\begin{equation*}
\boxed{\Delta = S(w^{var}) - S(w^{ref})}
\end{equation*}
\small Negative $\Delta$ = the alternate sequence is less plausible.
Youden's index on $-\Delta$: $J(t)=\mathrm{TPR}(t)-\mathrm{FPR}(t)$, $t^{\star}=\arg\max_t J(t)$;
\begin{equation*}
\mathrm{call} = \begin{cases}\text{Likely pathogenic}, & \Delta < \tau\\ \text{Likely benign}, & \text{otherwise}\end{cases}
\end{equation*}
\begin{equation*}
\boxed{\mathrm{conf}=\min\left(1,\frac{|\Delta-\tau|}{\sigma_{\mathrm{class}}}\right)}
\end{equation*}
\small $\sigma_{\mathrm{LOF}}\approx0.00151$, $\sigma_{\mathrm{FUNC}}\approx0.00090$; confidence is a distance from the threshold, capped at 1; not a calibrated probability. Indel paths reuse the SNV constants and flag \texttt{provisional}.
\end{column}

\begin{frame}{Methodology: Evaluation Design}
The deployed service was assessed along five axes.
\begin{enumerate}
\item \textbf{Calibration data:} re-read the workbook, recompute the class composition, confirm the reference bases against the shipped GRCh37 chr17 FASTA (itself validated against the live UCSC hg19 service).
\item \textbf{Reference envelope:} score the published predictors on the same 500 rows, under the same label definition and Youden procedure; stratified bootstrap over 2,000 resamples with a fixed seed.
\item \textbf{Window and assembly contract:} recompute both window rules over the 500 calibration positions; align GRCh37 and GRCh38 across BRCA1 with 681 conserved anchors.
\item \textbf{Validation conformance:} generate a 55-case matrix and run it against the real Pydantic model and the real Zod schema.
\item \textbf{Scoring contract:} exercise the deployed window/scoring functions with a stand-in scorer, so the arithmetic is checked without a graphics processor.
\end{enumerate}
Every number in this deck is reproducible from the evaluation scripts or is a source constant.
\end{frame}

\begin{frame}{Results: Window Contract}
\begin{itemize}\small
\item Calibration routine: window from $p-4096$ to $p+4096$, 8192 bases, variant at offset 4096.
\item Deployed endpoint: window from $p-1-4096$ to $p+4096$, 8193 bases, variant at offset 4096.
\item Corroborated on all 500 calibration positions: the two windows share their first 8192 bases; the endpoint adds one base at the 3' end; no clipping (no variant near a chromosome end).
\item The score difference is then a difference of means over 8193 vs 8192 terms; the deployed $\Delta$ distribution is not the one $\tau$ was fitted on.
\end{itemize}

\begin{frame}{Results: Scoring Contract Behaviour}
The deployed window and scoring functions were exercised with a stand-in scorer, to check the arithmetic of the contract without a graphics processor.
\begin{itemize}\small
\item A substitution at a BRCA1 position gave a score difference of $-1.03\times10^{-4}$ --- the same order of scale as the deployed threshold of $-9.18\times10^{-4}$.
\item A deletion of three bases scored sequences of 8193 and 8190 bases; an insertion of three bases scored 8193 and 8196 bases.
\item A null edit (alternate = reference) is accepted and yields a score difference of exactly zero; the call is benign and the confidence is $\min(1,|0-\tau|/\sigma_{\mathrm{FUNC}})=1.000000$.
\end{itemize}
\begin{equation*}
\min\left(1,\frac{|0-\tau|}{\sigma_{\mathrm{FUNC}}}\right)=1.000000
\end{equation*}
\begin{itemize}\small
\item The online path re-scores the reference window on every request (two forward passes); the offline routine de-duplicates reference windows (168 references for 500 variants).
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
The scoring contract of a deployed genomic language model service was formalized and measured; the calibration data reproduce the source assay composition; the first reproducible reference envelope for the slice was computed with bootstrap intervals; the window and assembly contracts were verified.
\begin{itemize}\small
\item 55 request cases: the two validation layers disagree on 6, and both accept 7 contract-violating cases; eight defects are reproducible from the scripts.
\item Evo2Serve is a usable interface to a model's opinion about a sequence, not yet a calibrated variant classifier.
\item Carrying the reference allele closes most of the defect set; its labels remain computational scores, not clinical explanations.
\end{itemize}
\end{frame}
